cmake_minimum_required(VERSION 3.16)
project(hx_platform_consumer LANGUAGES C CXX)
find_package(hx_c_platform 0.1 CONFIG REQUIRED)
enable_testing()
add_executable(consumer_c ../../examples/hello.c)
target_link_libraries(consumer_c PRIVATE hx_c_platform::hx_c_platform)
set_target_properties(consumer_c PROPERTIES
	C_STANDARD 99 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
add_executable(consumer_cpp consumer.cpp)
target_link_libraries(consumer_cpp PRIVATE hx_c_platform::hx_c_platform)
set_target_properties(consumer_cpp PROPERTIES
	CXX_STANDARD 11 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
if(MSVC)
	target_compile_options(consumer_c PRIVATE /utf-8)
	target_compile_options(consumer_cpp PRIVATE /utf-8)
endif()
add_test(NAME installed_c COMMAND consumer_c)
add_test(NAME installed_cpp COMMAND consumer_cpp)
add_executable(consumer_config ../../examples/configuration.c)
target_link_libraries(consumer_config PRIVATE hx_c_platform::hx_c_platform)
set_target_properties(consumer_config PROPERTIES
	C_STANDARD 99 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
if(MSVC)
	target_compile_options(consumer_config PRIVATE /utf-8)
endif()
add_test(NAME installed_config COMMAND consumer_config)
add_executable(consumer_containers ../../examples/containers.c)
target_link_libraries(consumer_containers PRIVATE
	hx_c_platform::hx_c_platform)
set_target_properties(consumer_containers PROPERTIES
	C_STANDARD 99 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
add_executable(consumer_tasks ../../examples/thread_pool.cpp)
target_link_libraries(consumer_tasks PRIVATE hx_c_platform::cpp)
set_target_properties(consumer_tasks PROPERTIES
	CXX_STANDARD 11 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
if(MSVC)
	target_compile_options(consumer_containers PRIVATE /utf-8)
	target_compile_options(consumer_tasks PRIVATE /utf-8)
endif()
add_test(NAME installed_containers COMMAND consumer_containers)
add_test(NAME installed_tasks COMMAND consumer_tasks)
